def make_windows(series, window):
    values = series.to_numpy(dtype=np.float32)

    X = []
    y = []
    target_dates = []

    for i in range(window, len(values)):
        block = values[i - window : i + 1]

        if not np.isfinite(block).all():
            continue

        X.append(block[:-1])
        y.append(block[-1])
        target_dates.append(series.index[i])

    X = np.asarray(X, dtype=np.float32)[..., None]

    y = np.asarray(y, dtype=np.float32).reshape(-1, 1)

    return X, y, pd.DatetimeIndex(target_dates)

class TemperatureGRU(nn.Module):
    def __init__(self, hidden_size=32):
        super().__init__()

        self.gru = nn.GRU(
            input_size=1,
            hidden_size=hidden_size,
            num_layers=1,
            batch_first=True
        )

        self.output = nn.Linear(hidden_size, 1)

    def forward(self, x):
        sequence, _ = self.gru(x)

        final_step = sequence[:, -1, :]

        return self.output(final_step)

def fit_gru(
    data,
    hidden_size=32,
    lr=0.001,
    batch_size=32,
    max_epochs=100,
    patience=10
):
    random.seed(SEED)
    np.random.seed(SEED)
    torch.manual_seed(SEED)

    model = TemperatureGRU(hidden_size=hidden_size)

    optimizer = torch.optim.Adam(
        model.parameters(),
        lr=lr
    )

    criterion = nn.MSELoss()

    X_train, y_train, _ = data["train"]
    X_val, y_val, _ = data["validation"]

    train_loader = DataLoader(
        TensorDataset(X_train, y_train),
        batch_size=batch_size,
        shuffle=True
    )

    best_loss = float("inf")
    best_state = None
    best_epoch = 0
    epochs_without_improvement = 0

    history = {
        "train": [],
        "validation": []
    }

    for epoch in range(max_epochs):
        model.train()
        total_train_loss = 0.0

        for X_batch, y_batch in train_loader:
            optimizer.zero_grad()

            predictions = model(X_batch)
            loss = criterion(predictions, y_batch)

            loss.backward()

            nn.utils.clip_grad_norm_(
                model.parameters(),
                max_norm=1.0
            )

            optimizer.step()

            total_train_loss += loss.item() * len(X_batch)

        train_loss = total_train_loss / len(X_train)

        model.eval()

        with torch.no_grad():
            val_predictions = model(X_val)
            val_loss = criterion(val_predictions, y_val).item()

        history["train"].append(train_loss)
        history["validation"].append(val_loss)

        if val_loss < best_loss:
            best_loss = val_loss
            best_state = copy.deepcopy(model.state_dict())
            best_epoch = epoch + 1
            epochs_without_improvement = 0
        else:
            epochs_without_improvement += 1

        if epoch == 0 or (epoch + 1) % 10 == 0:
            print(
                f"Epoch {epoch + 1:3d} | "
                f"Train MSE: {train_loss:.4f} | "
                f"Validation MSE: {val_loss:.4f}"
            )

        if epochs_without_improvement >= patience:
            print(f"Early stopping at epoch {epoch + 1}")
            break

    model.load_state_dict(best_state)
    model.eval()

    return model, history, best_epoch

def evaluate(model, split):
    X, y, dates = split

    model.eval()

    with torch.no_grad():
        predicted_z = model(X).numpy()

    predicted = scaler.inverse_transform(predicted_z).ravel()
    actual = scaler.inverse_transform(y.numpy()).ravel()

    naive = (
        data_melb_airport["MaxTemp"]
        .shift(1)
        .loc[dates]
        .to_numpy()
    )

    metrics = {
        "GRU MAE": mean_absolute_error(actual, predicted),
        "GRU RMSE": mean_squared_error(actual, predicted) ** 0.5,
        "Naive MAE": mean_absolute_error(actual, naive),
        "Naive RMSE": mean_squared_error(actual, naive) ** 0.5
    }

    forecasts = pd.DataFrame(
        {
            "Actual": actual,
            "GRU": predicted,
            "Naive": naive
        },
        index=dates
    )

    return metrics, forecasts
